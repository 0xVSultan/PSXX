# the identifiers (ui_ids.h) and the tables of the three ranges (ui_tabs_a.c, ui_tabs_b.c, ui_tabs_l.c) made from tools/strings/{en,hu}_<range>.txt
set(UIGEN ${CMAKE_BINARY_DIR}/gen)
file(GLOB UI_LISTS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tools/strings/*_?.txt)
add_custom_command(OUTPUT ${UIGEN}/ui_ids.h ${UIGEN}/ui_tabs_a.c ${UIGEN}/ui_tabs_b.c ${UIGEN}/ui_tabs_l.c
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/ui_ids.py gen --out ${UIGEN}
    DEPENDS ${UI_LISTS} ${CMAKE_SOURCE_DIR}/tools/ui_ids.py ${CMAKE_SOURCE_DIR}/tools/ui_strings.py)
add_custom_target(ui_gen DEPENDS ${UIGEN}/ui_ids.h ${UIGEN}/ui_tabs_a.c ${UIGEN}/ui_tabs_b.c ${UIGEN}/ui_tabs_l.c)       # a clean build makes the generated header before anything includes it
# the texts for every host program: the lookup, the generated tables and the registration of the English ones before main (the core asks for its texts by identifier, core/tr.h)
add_library(ui_objs OBJECT hal/psx/ui_text.c tests/ui_autoreg.c ${UIGEN}/ui_tabs_a.c ${UIGEN}/ui_tabs_b.c ${UIGEN}/ui_tabs_l.c)
target_include_directories(ui_objs PUBLIC hal/psx ${UIGEN})
target_compile_definitions(ui_objs PRIVATE UI_WITH_HU=1)     # the host programs have the Hungarian texts of the screens built in (the layout tests); on the console they come from the data card (a language of its own)

    if(NOT _n STREQUAL "gpu.h")
        configure_file(${CMAKE_SOURCE_DIR}/hal/psx/${_n} ${LUI}/${_n} COPYONLY)
    endif()
endforeach()
foreach(f gpu.h psxgpu.h)
    configure_file(${CMAKE_SOURCE_DIR}/tests/langui_stub/${f} ${LUI}/${f} COPYONLY)
endforeach()
